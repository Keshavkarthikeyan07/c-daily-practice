#include<stdio.h>
int main()
{
    int n,i;
    printf("Enter the number n:");
    scanf("%d",&n);
    for(i=1;i<=n;i++)
    if(i%3 == 0 && i%5 ==0){
        printf("FuzzBuzz\n");
    }
    else if(i%3 == 0){
        printf("Buzz\n");
    }
    else if(i%5 == 0){
        printf("Fuzz\n");
    }
    else{
        printf("%d\n",i);
    }
     return 0;
}
