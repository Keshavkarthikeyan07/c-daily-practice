int sumOddLengthSubarrays(int* arr, int arrSize)
{
    int total = 0;

    for (int i = 0; i < arrSize; i++)
    {
        for (int j = i; j < arrSize; j++)
        {
            if ((j - i + 1) % 2 != 0)
            {
                int sum = 0;

                for (int k = i; k <= j; k++)
                {
                    sum += arr[k];
                }

                total += sum;
            }
        }
    }

    return total;
}
